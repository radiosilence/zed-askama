================================================================================
If takes else if, elif and else branches
================================================================================
{% if a %}A{% else if b %}B{% elif c %}C{% else %}D{% endif %}

--------------------------------------------------------------------------------

(template
  (if_block
    (if_tag
      (identifier))
    (content)
    (else_if_clause
      (else_if_tag
        (identifier))
      (content))
    (else_if_clause
      (else_if_tag
        (identifier))
      (content))
    (else_clause
      (else_tag)
      (content))
    (endif_tag))
  (content))

================================================================================

--------------------------------------------------------------------------------
