================================================================================
Macro definitions take parameters with types and defaults
================================================================================

{% macro button(label, kind: &str, size = 2,) %}{{ label }}{% endmacro button %}

--------------------------------------------------------------------------------

(template
  (content)
  (macro_definition
    (macro_tag
      (identifier)
      (parameters
        (parameter
          (identifier))
        (parameter
          (identifier)
          (reference_type
            (type_identifier)))
        (parameter
          (identifier)
          (integer_literal))))
    (expression_tag
      (identifier))
    (endmacro_tag
      (identifier)))
  (content))

================================================================================
Macro definitions may omit the parameter list
================================================================================

{% macro m %}x{% endmacro %}

--------------------------------------------------------------------------------

(template
  (content)
  (macro_definition
    (macro_tag
      (identifier))
    (content)
    (endmacro_tag))
  (content))

================================================================================
Call blocks take scoped names, arguments and caller parameters
================================================================================

{% call(item) m::list(items, ordered = true) %}{{ item }}{% endcall %}

--------------------------------------------------------------------------------

(template
  (content)
  (call_block
    (call_tag
      (caller_parameters
        (identifier))
      (scoped_identifier
        (identifier)
        (identifier))
      (arguments
        (identifier)
        (named_argument
          (identifier)
          (boolean_literal))))
    (expression_tag
      (identifier))
    (endcall_tag))
  (content))

    (endcall_tag))
