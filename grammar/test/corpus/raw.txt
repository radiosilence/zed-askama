================================================================================
Raw content keeps tags as text
================================================================================
{% raw %}{{ not an expression }}{% if %}{% endraw %}

--------------------------------------------------------------------------------

(template
  (raw_block
    (raw_tag)
    (raw_content)
    (endraw_tag))
  (content))

================================================================================
Raw blocks take whitespace control on both tags
================================================================================
{%- raw -%} x {%- endraw -%}

--------------------------------------------------------------------------------

(template
  (raw_block
    (raw_tag)
    (raw_content)
    (endraw_tag))
  (content))

================================================================================
An empty raw block has no body
================================================================================
{% raw %}{% endraw %}

--------------------------------------------------------------------------------

(template
  (raw_block
    (raw_tag)
    (endraw_tag))
  (content))
